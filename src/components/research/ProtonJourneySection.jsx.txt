import { motion, useInView, useScroll, useTransform } from 'framer-motion';
import { useRef } from 'react';

  { pos: '5%',  label: 'H₂O enters', desc: 'Pure water is fed to anode', color: '#0EA5E9', side: 'top' },
  { pos: '20%', label: 'OER reaction', desc: '2H₂O → O₂ + 4H⁺ + 4e⁻', color: '#F59E0B', side: 'bottom' },
  { pos: '35%', label: 'O₂ evacuated', desc: 'Oxygen exits via flow field', color: '#EF4444', side: 'top' },
  { pos: '50%', label: 'H⁺ crosses Nafion', desc: 'Proton migrates through membrane', color: '#14B8A6', side: 'bottom' },

      <div className="absolute inset-0 opacity-[0.04]"

      {/* Orbs */}
      <div className="absolute top-20 left-1/4 w-96 h-96 rounded-full bg-ion/10 blur-3xl pointer-events-none" />
      <div className="absolute bottom-20 right-1/4 w-96 h-96 rounded-full bg-teal/10 blur-3xl pointer-events-none" />

      <div className="max-w-7xl mx-auto px-6">
        {/* Header */}
        <motion.div
          className="text-center mb-20"
          initial={{ opacity: 0, y: 24 }}
          animate={isInView ? { opacity: 1, y: 0 } : {}}
          transition={{ duration: 0.7, ease: [0.16, 1, 0.3, 1] }}
        >
          <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full bg-white/5 border border-white/10 mb-6">
            <div className="w-1.5 h-1.5 rounded-full bg-teal animate-pulse" />
            <span className="font-mono text-teal text-xs tracking-[0.2em] uppercase">The Proton Journey</span>
          </div>
          <h2 className="text-4xl md:text-5xl font-heading font-bold text-white mb-6">
            Follow a Single Proton
          </h2>
          <p className="text-lg text-white/40 max-w-xl mx-auto">
            From water molecule to pure hydrogen gas — visualised as you scroll.
          </p>
        </motion.div>

        {/* Scroll-driven path */}
        <div className="relative mb-20">
          {/* SVG path */}
          <svg className="w-full h-32 md:h-24" viewBox="0 0 1000 80" preserveAspectRatio="none">
            <motion.path
              d="M 0 40 Q 125 10 250 40 Q 375 70 500 40 Q 625 10 750 40 Q 875 70 1000 40"
              fill="none"
              stroke="rgba(14,165,233,0.15)"
              strokeWidth="2"
              strokeDasharray="6 4"
            />
            <motion.path
              d="M 0 40 Q 125 10 250 40 Q 375 70 500 40 Q 625 10 750 40 Q 875 70 1000 40"
              fill="none"
              stroke="#0EA5E9"
              strokeWidth="3"
              strokeLinecap="round"
              pathLength="1"
              style={{ pathLength: pathProgress }}
            />
          </svg>

          {/* Journey dots */}
          <div className="absolute inset-0">
            {JOURNEY.map((point, i) => (
              <motion.div
                key={i}
                className="absolute"
                style={{ left: point.pos, top: point.side === 'top' ? '-16px' : 'auto', bottom: point.side === 'bottom' ? '-16px' : 'auto' }}
                initial={{ opacity: 0, scale: 0 }}
                animate={isInView ? { opacity: 1, scale: 1 } : {}}
                transition={{ duration: 0.5, delay: 0.3 + i * 0.15, ease: [0.16, 1, 0.3, 1] }}
              >
                <div className="relative -translate-x-1/2 group">
                  <div className="w-4 h-4 rounded-full shadow-lg shadow-ion/30 border-2 border-cobalt cursor-pointer"
                    style={{ background: point.color }} />
                  <div className={`absolute ${point.side === 'top' ? 'bottom-6' : 'top-6'} left-1/2 -translate-x-1/2 w-max max-w-[140px] bg-white rounded-xl px-3 py-2 shadow-xl opacity-0 group-hover:opacity-100 transition-opacity duration-300 z-10`}>

            ))}

            { label: 'Proton travel time', value: '~ms', sub: 'Across Nafion membrane', color: '#14B8A6' },
            { label: 'H₂ energy density', value: '33.3', unit: 'kWh/kg', sub: 'vs 12.7 kWh/kg for CH₄', color: '#0EA5E9' },
            { label: 'H₂ purity output', value: '99.999', unit: '%', sub: 'After membrane dryer', color: '#8B5CF6' },
            { label: 'CO₂ emitted', value: '0', unit: 'g/kWh', sub: 'Zero-emission process', color: '#10B981' },
          ].map((stat, i) => (
            <motion.div
              key={stat.label}
