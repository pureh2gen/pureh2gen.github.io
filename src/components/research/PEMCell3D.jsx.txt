import { useRef, useEffect, useState } from 'react';
import * as THREE from 'three';
import { motion } from 'framer-motion';

const LAYERS = [
  { name: 'End Plate',      color: 0x334155, thickness: 0.14, label: 'End Plate',                      desc: 'Structural compression plate — holds the stack under 2–3 MPa clamping force. Contains inlet/outlet ports.' },
  { name: 'Bipolar Plate',  color: 0x0ea5e9, thickness: 0.10, label: 'Bipolar Plate',                  desc: 'Titanium plate with machined flow channels (serpentine or bio-inspired). Distributes water and collects current.' },
  { name: 'Cathode Gasket', color: 0xf43f5e, thickness: 0.04, label: 'Cathode Gasket',                 desc: 'PTFE/silicone seal on the cathode side. Prevents H₂ leakage and ensures correct compression of the GDL.' },
  { name: 'Cathode GDL',    color: 0x94a3b8, thickness: 0.06, label: 'Cathode GDL',                    desc: 'Gas Diffusion Layer — carbon fiber felt. Distributes H₂ flow, conducts electrons, provides mechanical support.' },
  { name: 'MEA',            color: 0x14b8a6, thickness: 0.06, label: 'MEA (Membrane Electrode Assembly)', desc: 'Core of the cell: Nafion® membrane + IrO₂ anode catalyst + Pt/C cathode catalyst. OER and HER occur here. Protons cross from anode to cathode.' },
  { name: 'Anode PTL',      color: 0xf59e0b, thickness: 0.06, label: 'Anode PTL',                      desc: 'Porous Transport Layer — titanium sintered felt (65–70% porosity). Evacuates O₂ bubbles and delivers water uniformly to the MEA.' },
  { name: 'Anode Gasket',   color: 0xa855f7, thickness: 0.04, label: 'Anode Gasket',                   desc: 'PTFE/silicone seal on the anode side. Contains water and O₂, prevents cross-leakage under operating pressure.' },
  { name: 'Bipolar Plate',  color: 0x0ea5e9, thickness: 0.10, label: 'Bipolar Plate',                  desc: 'Mirror bipolar plate on the anode side. Routes water in and O₂ out through its machined channel geometry.' },
  { name: 'End Plate',      color: 0x334155, thickness: 0.14, label: 'End Plate',                      desc: 'Closing compression plate. Provides uniform clamping force to maintain contact resistance and stack sealing.' },
];

// MEA is at index 4 — protons travel through it (Y axis in stack)
const MEA_INDEX = 4;

export default function PEMCell3D() {
  const mountRef = useRef(null);
  const animFrameRef = useRef(null);
  const isDraggingRef = useRef(false);
  const prevMouseRef = useRef({ x: 0, y: 0 });
  const rotationRef = useRef({ x: 0.3, y: 0.5 });
  const targetRotRef = useRef({ x: 0.3, y: 0.5 });
  const explodeRef = useRef(0);
  const targetExplodeRef = useRef(0);
  const hoveredLayerRef = useRef(null);
  const meshesRef = useRef([]);
  const particlesRef = useRef([]);
  const clockRef = useRef(new THREE.Clock());

  const [exploded, setExploded] = useState(false);
  const [hoveredLayer, setHoveredLayer] = useState(null);
  const [activeLayer, setActiveLayer] = useState(null);

  useEffect(() => {
    const el = mountRef.current;
    if (!el) return;

    const w = el.clientWidth;
    const h = el.clientHeight;

    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0xf8fafc);

    const camera = new THREE.PerspectiveCamera(45, w / h, 0.01, 100);
    camera.position.set(0, 0.4, 2.8);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(w, h);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.shadowMap.enabled = true;
    el.appendChild(renderer.domElement);

    // Lights
    scene.add(new THREE.AmbientLight(0xffffff, 0.7));
    const dir = new THREE.DirectionalLight(0xffffff, 1.2);
    dir.position.set(3, 5, 5);
    dir.castShadow = true;
    scene.add(dir);
    const fill = new THREE.DirectionalLight(0x90d5ff, 0.4);
    fill.position.set(-3, -2, -3);
    scene.add(fill);

    // Build layer meshes
    const totalThickness = LAYERS.reduce((s, l) => s + l.thickness, 0);
    let yOffset = -totalThickness / 2;
    const meshes = [];

    LAYERS.forEach((layer, i) => {
      const geo = new THREE.BoxGeometry(1.4, layer.thickness, 1.0);
      const mat = new THREE.MeshPhongMaterial({
        color: layer.color,
        transparent: true,
        opacity: 0.92,
        shininess: 60,
      });
      const mesh = new THREE.Mesh(geo, mat);
      const centerY = yOffset + layer.thickness / 2;
      mesh.position.y = centerY;
      mesh.userData = { baseY: centerY, index: i, layer };
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      meshes.push(mesh);
      yOffset += layer.thickness;
    });

    meshesRef.current = meshes;

    // Group for rotation — all objects go in here
    const group = new THREE.Group();
    meshes.forEach(m => group.add(m));
    scene.add(group);

    // Flow channel grooves on Bipolar Plates ONLY (indices 1 and 7)
    // Grooves run along Z axis, positioned on the inner face (bottom of BP1, top of BP7)
    [1, 7].forEach(idx => {
      const bp = meshes[idx];
      const faceY = idx === 1
        ? bp.userData.baseY - bp.userData.layer.thickness / 2  // bottom face of cathode BP
        : bp.userData.baseY + bp.userData.layer.thickness / 2; // top face of anode BP
      for (let g = 0; g < 7; g++) {
        const gGeo = new THREE.BoxGeometry(0.02, 0.008, 0.85);
        const gMat = new THREE.MeshPhongMaterial({ color: 0x0f172a, opacity: 0.6, transparent: true });
        const gMesh = new THREE.Mesh(gGeo, gMat);
        gMesh.position.set(-0.54 + g * 0.18, faceY, 0);
        group.add(gMesh);
      }
    });

    // --- PARTICLES ---
    // Protons (H⁺): teal spheres, travel from anode (bottom) through MEA upward to cathode
    // Electrons (e⁻): purple spheres, travel externally along the sides (X direction)
    const protonGeo = new THREE.SphereGeometry(0.018, 8, 8);
    const protonMat = new THREE.MeshPhongMaterial({ color: 0x14b8a6, emissive: 0x14b8a6, emissiveIntensity: 0.5 });
    const electronGeo = new THREE.SphereGeometry(0.014, 8, 8);
    const electronMat = new THREE.MeshPhongMaterial({ color: 0xa855f7, emissive: 0xa855f7, emissiveIntensity: 0.5 });

    const particles = [];

    // 6 protons — start below MEA (anode side), travel upward through it to cathode
    for (let p = 0; p < 6; p++) {
      const mesh = new THREE.Mesh(protonGeo, protonMat.clone());
      mesh.visible = false;
      group.add(mesh);
      particles.push({
        mesh,
        type: 'proton',
        phase: p / 6, // staggered start
        xOffset: -0.35 + (p % 3) * 0.35,
        zOffset: p < 3 ? -0.15 : 0.15,
      });
    }

    // 6 electrons — travel along the outer edge of the stack (externally)
    for (let e = 0; e < 6; e++) {
      const mesh = new THREE.Mesh(electronGeo, electronMat.clone());
      mesh.visible = false;
      group.add(mesh);
      particles.push({
        mesh,
        type: 'electron',
        phase: e / 6,
        side: e < 3 ? 1 : -1, // left or right side
        zOffset: -0.2 + (e % 3) * 0.2,
      });
    }

    particlesRef.current = particles;

    // Raycaster
    const raycaster = new THREE.Raycaster();
    const mouse = new THREE.Vector2();

    const getHit = (e) => {
      const rect = el.getBoundingClientRect();
      mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
      raycaster.setFromCamera(mouse, camera);
      const hits = raycaster.intersectObjects(meshes);
      return hits.length > 0 ? hits[0].object : null;
    };

    const onMouseMove = (e) => {
      if (isDraggingRef.current) {
        const dx = e.clientX - prevMouseRef.current.x;
        const dy = e.clientY - prevMouseRef.current.y;
        targetRotRef.current.y += dx * 0.008;
        targetRotRef.current.x += dy * 0.008;
        targetRotRef.current.x = Math.max(-0.8, Math.min(0.8, targetRotRef.current.x));
        prevMouseRef.current = { x: e.clientX, y: e.clientY };
      }
      const hit = getHit(e);
      const idx = hit ? hit.userData.index : null;
      hoveredLayerRef.current = idx;
      setHoveredLayer(idx);
    };
    const onMouseDown = (e) => {
      isDraggingRef.current = true;
      prevMouseRef.current = { x: e.clientX, y: e.clientY };
    };
    const onMouseUp = () => { isDraggingRef.current = false; };
    const onClick = (e) => {
      const hit = getHit(e);
      setActiveLayer(hit ? hit.userData.index : null);
    };
    let lastTouch = null;
    const onTouchStart = (e) => { lastTouch = e.touches[0]; };
    const onTouchMove = (e) => {
      if (!lastTouch) return;
      const t = e.touches[0];
      targetRotRef.current.y += (t.clientX - lastTouch.clientX) * 0.01;
      targetRotRef.current.x += (t.clientY - lastTouch.clientY) * 0.01;
      targetRotRef.current.x = Math.max(-0.8, Math.min(0.8, targetRotRef.current.x));
      lastTouch = t;
    };

    el.addEventListener('mousemove', onMouseMove);
    el.addEventListener('mousedown', onMouseDown);
    window.addEventListener('mouseup', onMouseUp);
    el.addEventListener('click', onClick);
    el.addEventListener('touchstart', onTouchStart, { passive: true });
    el.addEventListener('touchmove', onTouchMove, { passive: true });

    const animate = () => {
      animFrameRef.current = requestAnimationFrame(animate);
      const t = clockRef.current.getElapsedTime();

      // Smooth rotation
      rotationRef.current.x += (targetRotRef.current.x - rotationRef.current.x) * 0.08;
      rotationRef.current.y += (targetRotRef.current.y - rotationRef.current.y) * 0.08;
      group.rotation.x = rotationRef.current.x;
      group.rotation.y = rotationRef.current.y;
      if (!isDraggingRef.current) targetRotRef.current.y += 0.003;

      // Explode animation
      explodeRef.current += (targetExplodeRef.current - explodeRef.current) * 0.06;
      const exp = explodeRef.current;

      const totalT = LAYERS.reduce((s, l) => s + l.thickness, 0);
      let baseY = -totalT / 2;
      const layerCenterYs = [];
      meshes.forEach((m, i) => {
        const layer = LAYERS[i];
        const center = baseY + layer.thickness / 2;
        const explodeY = center + (i - LAYERS.length / 2) * exp * 0.28;
        m.position.y = explodeY;
        layerCenterYs.push(explodeY);
        const isHov = hoveredLayerRef.current === i;
        m.material.emissive.setHex(isHov ? 0x334155 : 0x000000);
        m.material.emissiveIntensity = isHov ? 0.15 : 0;
        baseY += layer.thickness;
      });

      // Particle animation — only meaningful when exploded (exp > 0.3)
      const meaY = layerCenterYs[MEA_INDEX] ?? 0;
      const anodeBottomY = layerCenterYs[5] ?? -0.1; // Anode PTL
      const cathodeTopY = layerCenterYs[3] ?? 0.1;   // Cathode GDL
      const stackHalfH = totalT / 2 + exp * 0.28 * LAYERS.length / 2;

      particles.forEach((p) => {
        const speed = 0.6;
        const cycle = ((t * speed + p.phase) % 1);

        if (p.type === 'proton') {
          // Travel from anode side (below MEA) upward through MEA to cathode side
          const startY = anodeBottomY - 0.05;
          const endY = cathodeTopY + 0.05;
          p.mesh.position.set(p.xOffset, startY + (endY - startY) * cycle, p.zOffset);
          // Fade in/out at boundaries
          const alpha = Math.sin(cycle * Math.PI);
          p.mesh.material.opacity = exp * alpha * 0.9;
          p.mesh.material.transparent = true;
          p.mesh.visible = exp > 0.2;
          // Pulse scale near MEA
          const distToMEA = Math.abs(p.mesh.position.y - meaY);
          const pulse = 1 + 0.4 * Math.max(0, 1 - distToMEA / 0.1);
          p.mesh.scale.setScalar(pulse);
        } else {
          // Electrons travel externally: down the side from cathode top to anode bottom
          const startY = cathodeTopY + 0.05;
          const endY = anodeBottomY - 0.05;
          const xPos = p.side * 0.82; // outside the stack
          p.mesh.position.set(xPos, startY + (endY - startY) * cycle, p.zOffset);
          const alpha = Math.sin(cycle * Math.PI);
          p.mesh.material.opacity = exp * alpha * 0.85;
          p.mesh.material.transparent = true;
          p.mesh.visible = exp > 0.2;
          p.mesh.scale.setScalar(1);
        }
      });

      renderer.render(scene, camera);
    };
    animate();

    const onResize = () => {
      const w2 = el.clientWidth, h2 = el.clientHeight;
      camera.aspect = w2 / h2;
      camera.updateProjectionMatrix();
      renderer.setSize(w2, h2);
    };
    window.addEventListener('resize', onResize);

    return () => {
      cancelAnimationFrame(animFrameRef.current);
      el.removeEventListener('mousemove', onMouseMove);
      el.removeEventListener('mousedown', onMouseDown);
      window.removeEventListener('mouseup', onMouseUp);
      el.removeEventListener('click', onClick);
      el.removeEventListener('touchstart', onTouchStart);
      el.removeEventListener('touchmove', onTouchMove);
      window.removeEventListener('resize', onResize);
      renderer.dispose();
      if (el.contains(renderer.domElement)) el.removeChild(renderer.domElement);
    };
  }, []);

  const toggleExplode = () => {
    const next = !exploded;
    setExploded(next);
    targetExplodeRef.current = next ? 1 : 0;
    if (!next) setActiveLayer(null);
  };

  const activeInfo = activeLayer !== null ? LAYERS[activeLayer] : hoveredLayer !== null ? LAYERS[hoveredLayer] : null;
  const displayIndex = activeLayer !== null ? activeLayer : hoveredLayer;

  return (
    <div className="relative w-full h-full">
      <div ref={mountRef} className="w-full h-full cursor-grab active:cursor-grabbing rounded-3xl overflow-hidden" />

      {/* Controls */}
      <div className="absolute bottom-4 left-1/2 -translate-x-1/2 flex items-center gap-3">
        <button
          onClick={toggleExplode}
          className={`px-5 py-2.5 rounded-full text-xs font-semibold font-mono transition-all duration-400 shadow-lg ${
            exploded ? 'bg-ion text-white shadow-ion/30' : 'bg-white border border-slate/15 text-slate/70 hover:border-ion/30 hover:text-ion'
          }`}
        >
          {exploded ? '⊙ Collapse' : '⊕ Exploded View'}
        </button>
        <span className="text-[10px] text-slate/30 font-mono hidden sm:block">Drag to rotate</span>
      </div>

      {/* Particle legend — only shown when exploded */}
      {exploded && (
        <motion.div
          initial={{ opacity: 0, y: -6 }}
          animate={{ opacity: 1, y: 0 }}
          className="absolute bottom-14 left-1/2 -translate-x-1/2 flex items-center gap-4 bg-white/90 backdrop-blur-sm rounded-full px-5 py-2 border border-slate/10 shadow-md"
        >
          <div className="flex items-center gap-1.5">
            <div className="w-2.5 h-2.5 rounded-full bg-teal-400 shadow shadow-teal-300" />
            <span className="text-[10px] font-mono text-slate/60">H⁺ protons (through MEA)</span>
          </div>
          <div className="w-px h-3 bg-slate/15" />
          <div className="flex items-center gap-1.5">
            <div className="w-2.5 h-2.5 rounded-full bg-violet-400 shadow shadow-violet-300" />
            <span className="text-[10px] font-mono text-slate/60">e⁻ electrons (external circuit)</span>
          </div>
        </motion.div>
      )}

      {/* Layer info tooltip */}
      {activeInfo && displayIndex !== null && (
        <motion.div
          key={displayIndex}
          initial={{ opacity: 0, y: 6 }}
          animate={{ opacity: 1, y: 0 }}
          className="absolute top-4 left-4 right-4 max-w-xs bg-white rounded-2xl border border-slate/10 shadow-xl p-5"
        >
          <div className="flex items-center gap-2 mb-2">
            <div className="w-3 h-3 rounded-full" style={{ background: `#${LAYERS[displayIndex].color.toString(16).padStart(6, '0')}` }} />
            <p className="font-heading font-bold text-slate text-sm">{activeInfo.label}</p>
          </div>
          <p className="text-xs text-slate/60 leading-relaxed">{activeInfo.desc}</p>
          <p className="text-[10px] text-slate/30 font-mono mt-2">Click layer to pin · Drag to rotate</p>
        </motion.div>
      )}

      {/* Layer legend */}
      {exploded && (
        <motion.div
          initial={{ opacity: 0, x: 10 }}
          animate={{ opacity: 1, x: 0 }}
          className="absolute right-4 top-4 bg-white rounded-2xl border border-slate/10 shadow-lg p-4 space-y-2 max-w-[160px]"
        >
          {LAYERS.map((l, i) => (
            <button
              key={i}
              onClick={() => setActiveLayer(i === activeLayer ? null : i)}
              className={`flex items-center gap-2 w-full text-left transition-colors ${activeLayer === i ? 'opacity-100' : 'opacity-60 hover:opacity-90'}`}
            >
              <div className="w-2.5 h-2.5 rounded-sm flex-shrink-0" style={{ background: `#${l.color.toString(16).padStart(6, '0')}` }} />
              <span className="text-[10px] font-mono text-slate truncate">{l.label}</span>
            </button>
          ))}
        </motion.div>
      )}
    </div>
  );
}
