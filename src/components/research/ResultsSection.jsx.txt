const CustomBarTooltip = ({ active, payload }) => {
  if (!active || !payload?.length) return null;
  const d = payload[0].payload;
  return (
    <div className="bg-white border border-slate/10 rounded-xl px-4 py-3 shadow-xl">
      <p className="font-mono text-slate/50 text-[10px] uppercase tracking-wider">{d.name}</p>
      <p className="font-mono text-slate text-lg font-bold">{d.value} <span className="text-sm text-slate/40">Pa</span></p>
    </div>
  );
};

const CustomLineTooltip = ({ active, payload, label }) => {
  if (!active || !payload?.length) return null;
  return (
    <div className="bg-white border border-slate/10 rounded-xl px-4 py-3 shadow-xl">
      <p className="font-mono text-slate/50 text-[10px] uppercase tracking-wider">j = {label} A/cm²</p>
      <p className="font-mono text-ion text-lg font-bold">{payload[0].value.toFixed(2)} <span className="text-sm text-slate/40">V</span></p>
    </div>
  );
};

const CustomBar = (props) => {
  const { x, y, width, height, name } = props;
  return <rect x={x} y={y} width={width} height={height} fill={COLORS[name]} rx={8} ry={8} />;
};

const KPI = ({ metric, label, desc, delay, color }) => {
  const ref = useRef(null);
  const isInView = useInView(ref, { once: true });
  return (
    <motion.div
      ref={ref}
      className="relative p-8 rounded-3xl border border-slate/8 bg-white card-hover overflow-hidden"
      initial={{ opacity: 0, y: 24 }}
      animate={isInView ? { opacity: 1, y: 0 } : {}}
      transition={{ duration: 0.6, delay, ease: [0.16, 1, 0.3, 1] }}
    >
      <div className={`absolute top-0 right-0 w-32 h-32 rounded-full blur-3xl opacity-10`} style={{ background: color }} />
      <p className={`font-mono text-5xl font-extrabold mb-3`} style={{ color }}>{metric}</p>
      <p className="font-heading font-semibold text-slate mb-1">{label}</p>
      <p className="text-sm text-slate/40">{desc}</p>
    </motion.div>
  );
};
